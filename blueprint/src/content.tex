\chapter{Analytic Degree, Theorem 1.1, and Core Reduction Lemmas}

This chapter establishes the degree-theoretic foundations (\S5 of \texttt{arXiv:2606.30403}),
bridges the main theorem of \texttt{arXiv:2606.30403} (Theorem~1.1) and Remark~1.15 to the
upstream \texttt{RCDEntropyDegree.RCD} library, and isolates the reusable strictness and metric-doubling
meta-lemmas in \texttt{AlexandrovConsequences/Foundation.lean}.

\begin{definition}[Analytic Degree and Mitsuishi Comparison Bridge]
\label{def_analytic_degree}
\lean{AlexandrovConsequences.analyticDegree, AlexandrovConsequences.MitsuishiDegreeBridge}
\leanok
Let $(X, \mathsf{d}, \mu_X)$ be a finite-measure space and $Y$ a closed oriented Riemannian
$N$-manifold with volume $\Vol(Y) > 0$. The \emph{analytic degree} of a map $f : X \to Y$ with
signed metric Jacobian $\mathrm{sJac}_f$ is
\[
  \deg_{\mathrm{an}}(f) := \frac{1}{\Vol(Y)} \int_X \mathrm{sJac}_f(x)\,d\mu_X(x).
\]
A \emph{Mitsuishi degree bridge} packages $Y$ together with a fundamental current, singular
homology degree $\deg_{\mathrm{top}}(f)$, current pushforward degree $\deg_{\mathrm{curr}}(f)$,
and the naturality identity $\deg_{\mathrm{curr}}(f) = \deg_{\mathrm{top}}(f)$ induced by
Mitsuishi's comparison isomorphism $\Psi : H_N^{\mathbf{I}}(X) \xrightarrow{\sim} H_N(X;\Z)$.
\end{definition}

\begin{lemma}[Analytic Degree Integral Bound]
\label{lem_analytic_degree_integral_bound}
\lean{AlexandrovConsequences.abs_analyticDegree_mul_vol_le_integral_abs}
\leanok
\uses{def_analytic_degree}
For any integrable signed metric Jacobian $\mathrm{sJac}_f$ and $\Vol(Y) > 0$,
$|\deg_{\mathrm{an}}(f)|\,\Vol(Y) \le \int_X |\mathrm{sJac}_f(x)|\,d\mu_X(x)$.
\end{lemma}
\begin{proof}\leanok
Cancel $\Vol(Y)^{-1} \cdot \Vol(Y) = 1$ using $\Vol(Y) > 0$ and apply the triangle inequality
for the Bochner integral.
\end{proof}

\begin{theorem}[Analytic Degree Equals Topological Degree on Class $\mathcal{S}$ Spaces]
\label{thm_analytic_eq_topological_degree}
\lean{AlexandrovConsequences.analytic_eq_topological_degree, AlexandrovConsequences.ClassSSpace, AlexandrovConsequences.s_space_analytic_eq_topological_degree}
\leanok
\uses{def_analytic_degree, lem_analytic_degree_integral_bound}
On any class $\mathcal{S}$ space $X$ (compact orientable $N$-metric space with
$\mathcal{H}^{N-1}(\mathcal{S}_X) = 0$ and Mitsuishi comparison isomorphism $\Psi$), every
Lipschitz map $f : X \to Y$ satisfies Kirchheim's signed area formula and hence
\[
  \deg_{\mathrm{an}}(f) = \deg_{\mathrm{top}}(f).
\]
\end{theorem}
\begin{proof}\leanok
Substituting Kirchheim's signed area identity $\int_X \mathrm{sJac}_f\,d\mathcal{H}^N =
\deg_{\mathrm{curr}}(f)\,\mathcal{H}^N(Y)$ into $\deg_{\mathrm{an}}(f)$ cancels $\mathcal{H}^N(Y)$,
and Mitsuishi naturality yields $\deg_{\mathrm{top}}(f)$.
\end{proof}

\begin{theorem}[Theorem 1.1: Alexandrov Entropy-Degree Theorem and Smoothness Rigidity]
\label{thm_alexandrov_entropy_degree}
\lean{AlexandrovConsequences.alexandrov_entropy_degree_inequality_continuous, AlexandrovConsequences.alexandrov_entropy_degree_rigidity}
\leanok
\uses{thm_analytic_eq_topological_degree}
Let $X$ be a closed orientable $N$-dimensional Alexandrov space without boundary
($\mathrm{curv} \ge \kappa$) and $f : X \to Y$ a continuous map into a closed orientable
negatively curved locally symmetric $N$-manifold $Y$. Via Lipschitz approximation
$f_{\mathrm{lip}} \simeq f$,
\[
  |\deg(f)| \, h(\widetilde{Y})^N \, \mathcal{H}^N(Y) \le h(\overline{X})^N \, \mathcal{H}^N(X).
\]
Furthermore, if equality holds together with $|\deg(f)| = [\pi_1(Y) : f_*\pi_1(X)] \ge 1$, then
the rescaled limit map $\hat{F}_\infty : (X, c\,\mathsf{d}_X) \to \widehat{Y}$
($c = h(\overline{X})/h(\widetilde{Y}) > 0$) is a surjective global homothety onto the
degree-$|\deg(f)|$ cover $\widehat{Y}$, and hence $\mathcal{S}(X) = \emptyset$.
\end{theorem}
\begin{proof}\leanok
Homotopy invariance equates $\deg(f) = \deg(f_{\mathrm{lip}})$ in
\texttt{RCDEntropyDegree.RCD.alexandrov\_entropy\_degree\_inequality}, and equality rigidity transfers smoothness
from $\widehat{Y}$ via \texttt{RCDEntropyDegree.RCD.rcd\_entropy\_degree\_equality\_rigidity}.
\end{proof}

\begin{lemma}[Remark 1.15: Necessity of the Index Condition and Connected-Sum Counterexample]
\label{lem_remark_index_necessity}
\lean{AlexandrovConsequences.abs_degree_eq_index_iff_lift_degree_one, AlexandrovConsequences.connected_sum_rigidity_counterexample}
\leanok
\uses{thm_alexandrov_entropy_degree}
In the covering factorization $|\deg(f)| = |\deg(\hat{f})| \cdot [\pi_1(Y) : f_*\pi_1(X)]$ with
$\deg(f) \neq 0$, we have $|\deg(f)| = [\pi_1(Y) : f_*\pi_1(X)] \iff |\deg(\hat{f})| = 1$.
For the pinch map $Y \# Y \to Y \vee Y \to Y$ ($b_1(Y) \ge 1$), $|\deg(f)| = 2 > 1 = [\pi_1(Y):f_*\pi_1(X)]$
and $b_1(Y \# Y) = 2 b_1(Y) \neq b_1(Y)$.
\end{lemma}
\begin{proof}\leanok
Cancel the positive integer index $[\pi_1(Y) : f_*\pi_1(X)] \ge 1$ in the degree factorization, and
compare $2 b_1(Y) \neq b_1(Y)$ for $b_1(Y) \ge 1$.
\end{proof}

\begin{lemma}[Strictness, Entropy Bounds, and Metric Doubling Cancellation]
\label{lem_strictness_and_doubling}
\lean{AlexandrovConsequences.strict_entropy_volume_of_singularSet_nonempty, AlexandrovConsequences.upper_entropy_bound_volume_inequality, AlexandrovConsequences.same_entropy_volume_degree_bound, AlexandrovConsequences.same_entropy_strict_volume_degree_bound, AlexandrovConsequences.metric_doubling_volume_degree_bound}
\leanok
\uses{thm_alexandrov_entropy_degree}
\begin{enumerate}
\item Whenever $\mathcal{S}_X \neq \emptyset$, $|\deg(f)| = [\pi_1(Y):f_*\pi_1(X)] \ge 1$, and
equality forces $\mathcal{S}_X = \emptyset$, the entropy-volume inequality is strict ($<$).
\item Whenever $0 \le h(\overline{X}) \le h_0$ ($h_0 > 0$) and $h(\widetilde{Y}) = h_0$, Theorem~1.1
yields $|\deg(f)|\Vol(Y) \le \mathcal{H}^N(X)$ (and $<$ when strict).
\item Under Perelman/Petrunin doubling $\mathcal{H}^N(DZ) = 2\mathcal{H}^N(Z)$ and
$\mathcal{H}^N(DY) = 2\mathcal{H}^N(Y)$, cancelling $2 h_0^N > 0$ gives $|k|\mathcal{H}^N(Y) \le \mathcal{H}^N(Z)$.
\end{enumerate}
\end{lemma}
\begin{proof}\leanok
Any equality contradicts $\mathcal{S}_X \neq \emptyset$; monotonicity of $x \mapsto x^N$ on
$[0,\infty)$ and positivity of $h_0^N$ and $2 h_0^N$ give the volume and doubling bounds.
\end{proof}

\chapter{Branched Coverings and Einstein 4-Orbifold Obstructions}

\begin{definition}[Cyclic Branched Cover Cone Angle and Density Drop]
\label{def_branched_cone_angle}
\lean{AlexandrovConsequences.branchedConeAngle, AlexandrovConsequences.branchedCoverConeAngle_bounds, AlexandrovConsequences.not_mem_regularSet_of_coneAngle_lt_two_pi, AlexandrovConsequences.branchedQuotient_singularSet_nonempty}
\leanok
For a $d$-fold cyclic branched cover $M_d \to M$ ($d \ge 2$), the quotient $X = M_d / \Z_d$
carries normal cone angle $\alpha = 2\pi / d \le \pi < 2\pi$ along the branch locus $\Sigma$,
giving Bishop--Gromov density $\theta_N(x_0) = \alpha/(2\pi) < 1$ and hence
$\mathcal{S}_X \neq \emptyset$.
\end{definition}

\begin{corollary}[Corollary 1.2: Branched Coverings and Hamenst\"adt's Question]
\label{cor_hamenstadt_gen}
\lean{AlexandrovConsequences.hamenstadt_quotient_strict_volume_gap, AlexandrovConsequences.hamenstadt_branched_cover_volume_gap}
\leanok
\uses{def_branched_cone_angle, lem_strictness_and_doubling}
Let $M$ be a closed hyperbolic $N$-manifold ($N \ge 3$) and $X = M_d / \Z_d$ ($d \ge 2$) the
orbifold quotient of a branched cover along a nonempty codimension-$2$ locus $\Sigma$ carrying a
negatively curved metric ($h(\overline{X}) \le N - 1$). Then
\[
  \mathcal{H}^N(M_d / \Z_d) > \Vol(M) \qquad \text{and} \qquad \Vol(M_d) > d \cdot \Vol(M).
\]
\end{corollary}
\begin{proof}\leanok
Since $\alpha = 2\pi/d < 2\pi$ forces $\mathcal{S}_X \neq \emptyset$, the strictness and
matched-entropy lemmas give $\mathcal{H}^N(X) > \Vol(M)$, and multiplying by $d \ge 2$ yields
$\Vol(M_d) = d\,\mathcal{H}^N(X) > d\,\Vol(M)$.
\end{proof}

\begin{definition}[Satake Chern--Gauss--Bonnet and Einstein 4-Orbifold Package]
\label{def_satake_einstein_data}
\lean{AlexandrovConsequences.fourManifoldGaussBonnetConst, AlexandrovConsequences.fourManifoldGaussBonnetConst_pos, AlexandrovConsequences.SatakeEinsteinFourOrbifoldData}
\leanok
With Chern--Gauss--Bonnet constant $C_4 = 4\pi^2/3 > 0$, a $4$-orbifold $(X, g_E)$ with
$\mathrm{Ric}(g_E) = -3 g_E$ satisfies $h(\overline{X}) \le 3$ and Satake's inequality
$\Vol(X, g_E) \le \frac{4\pi^2}{3}\chi_{\mathrm{orb}}(X)$, while a hyperbolic $4$-manifold $Y$
satisfies $h(\widetilde{Y}) = 3$ and $\Vol(Y) = \frac{4\pi^2}{3}\chi(Y)$.
\end{definition}

\begin{corollary}[Corollary 1.3(a)--(b): Einstein 4-Orbifold Euler Obstructions]
\label{cor_einstein_euler_obstruction}
\lean{AlexandrovConsequences.einstein_four_orbifold_euler_obstruction, AlexandrovConsequences.einstein_four_orbifold_strict_euler_obstruction}
\leanok
\uses{def_satake_einstein_data, lem_strictness_and_doubling}
Under a continuous map $f : X \to Y$ as above, $\chi_{\mathrm{orb}}(X) \ge |\deg(f)|\,\chi(Y)$.
Moreover, if $\mathcal{S}_X \neq \emptyset$ and $|\deg(f)| = [\pi_1(Y):f_*\pi_1(X)] \ge 1$
(forcing $\deg(f) \neq 0$), then strict inequality holds: $\chi_{\mathrm{orb}}(X) > |\deg(f)|\,\chi(Y)$.
\end{corollary}
\begin{proof}\leanok
Cancel $3^4 = 81$ via matched entropy $h(\overline{X}) \le 3$ and cancel $4\pi^2/3 > 0$ in the
Satake Chern--Gauss--Bonnet inequalities; nonempty $\mathcal{S}_X$ upgrades $\le$ to $<$.
\end{proof}

\begin{corollary}[Corollary 1.3(c): Gromov--Thurston Branch-Locus Euler Sign]
\label{cor_gromov_thurston_euler_neg}
\lean{AlexandrovConsequences.gromov_thurston_branch_locus_euler_neg}
\leanok
\uses{def_branched_cone_angle, cor_einstein_euler_obstruction}
For $X = M_d / \Z_d$ ($d \ge 2$) branched along a closed surface $\Sigma \subset M$, any Einstein
metric $\mathrm{Ric}(g_E) = -3 g_E$ on $X$ forces $\chi(\Sigma) < 0$.
\end{corollary}
\begin{proof}\leanok
Combining the orbifold Riemann--Hurwitz identity $\chi_{\mathrm{orb}}(X) = \chi(M) - (1 - 1/d)\chi(\Sigma)$
with $\chi_{\mathrm{orb}}(X) > \chi(M)$ and $1 - 1/d > 0$ yields $\chi(\Sigma) < 0$.
\end{proof}

\begin{theorem}[Theorem 1.10: Boundary Volume Inequality and Rigidity]
\label{thm_boundary_rigidity}
\lean{AlexandrovConsequences.alexandrov_boundary_volume_inequality, AlexandrovConsequences.alexandrov_boundary_volume_rigidity}
\leanok
\uses{cor_manifold_boundary, thm_alexandrov_entropy_degree}

mapped with degree $k \neq 0$ to a compact hyperbolic $N$-manifold $Y$ with totally geodesic
boundary, $\Vol(X) \ge |k|\,\Vol(Y)$. Moreover, if $\Vol(X) = |k|\,\Vol(Y)$ and

\end{theorem}
\begin{proof}\leanok
Volume equality $\Vol(DX) = |k|\,\Vol(DY) > 0$ together with Theorem~1.1 sandwiches
$(N-1)^N \le h(\overline{DX})^N \le (N-1)^N$, forcing $h(\overline{DX}) = N - 1$ by strict
monotonicity of $x \mapsto x^N$ on $[0, \infty)$ and triggering unrescaled ($c = 1$) rigidity.
\end{proof}

\begin{lemma}[Reviron Entropy Lower-Semicontinuity Product Limit]
\label{lem_reviron_entropy_lsc}
\lean{AlexandrovConsequences.limit_entropy_pow_le_of_eventually_add_eps}
\leanok
If $\Vol(M_i) \to V_X > 0$, $h(\widetilde{M}_i)^N \Vol(M_i) \to L$, and
$\forall \varepsilon > 0, \forall^\infty i,\, h_{\mathrm{lim}} \le h(\widetilde{M}_i) + \varepsilon$,
then $h_{\mathrm{lim}}^N V_X \le L$.
\end{lemma}
\begin{proof}\leanok
Since $V_X > 0$, $h(\widetilde{M}_i)^N \to L / V_X$. For each $\varepsilon > 0$, eventually
$(\max(h_{\mathrm{lim}} - \varepsilon, 0))^N \le h(\widetilde{M}_i)^N$, so taking $i \to \infty$
and then $\varepsilon \to 0^+$ yields $h_{\mathrm{lim}}^N \le L / V_X$.
\end{proof}

\begin{corollary}[Corollary 1.11: Gromov--Hausdorff Stability of Hyperbolic Volume]

\lean{AlexandrovConsequences.gh_hyperbolic_volume_stability}
\leanok

transfers $|\deg| = \mathrm{index} = k$, then $\Vol(X) = k\,\Vol(Y)$, $h(\widetilde{X}) = N - 1$,
$\mathcal{S}_X = \emptyset$, and $X$ is globally isometric to the smooth degree-$k$ hyperbolic cover
of $Y$ ($Y$ itself when $k = 1$).
\end{corollary}
\begin{proof}\leanok
Uniqueness of Hausdorff limits equates $\Vol(X) = k\,\Vol(Y)$; Perelman stability transfers
degree/index $k$; and Theorem~1.1 squeezes $h(\widetilde{X}) = N - 1$, triggering unrescaled rigidity.

\begin{lemma}[Selberg Finite Cover Cone-Angle Unwrapping]
\label{lem_selberg_cone_unwrap}
\lean{AlexandrovConsequences.unwrapped_cone_angle_le_two_pi_iff, AlexandrovConsequences.unwrapped_cone_angle_lt_two_pi_iff}
\leanok
Along a torsion-free Selberg finite cover $p : M \to \mathcal{O}$ with local isotropy orders
$m_i \ge 1$, the lifted cone angle satisfies $m_i \alpha_i \le 2\pi \iff \alpha_i \le 2\pi / m_i$
(and similarly for $<$).
\end{lemma}
\begin{proof}\leanok
Multiply or divide both sides by $(m_i : \R) > 0$.
\end{proof}
